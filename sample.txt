@Override
public void onTestStart(ITestResult result) {
    // --- Determine parent (class-level) name ---
    Class<?> clazz = result.getTestClass().getRealClass();
    ReportName reportNameAnnotation = clazz.getAnnotation(ReportName.class);
    String parentName = (reportNameAnnotation != null && !reportNameAnnotation.value().isEmpty())
            ? reportNameAnnotation.value()
            : clazz.getSimpleName();

    // Create parent node for the class (only once, thread-safe across parallel threads)
    ExtentTest parent = classTests.computeIfAbsent(parentName,
            name -> ExtentManager.getInstance().createTest(name));

    // --- Determine child (method-level) name ---
    String methodName = result.getMethod().getMethodName();
    String description = result.getMethod().getDescription();
    String childName = (description != null && !description.isEmpty())
            ? methodName + " — " + description
            : methodName;

    // Create child node under the class parent
    ExtentTest child = parent.createNode(childName);

    // Assign TestNG groups as categories (shown as filter chips in SparkReport)
    String[] groups = result.getMethod().getGroups();
    if (groups != null && groups.length > 0) {
        child.assignCategory(groups);
    }

    // Store in ThreadLocal so @BeforeMethod, test method, and lifecycle methods can access it
    ExtentManager.setTest(child);
}
