package com.selenium.automation.annotations;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

@Retention(RetentionPolicy.RUNTIME)
@Target(ElementType.TYPE)
public @interface ReportName {
    String value();
}



@ReportName("Claims Page API Tests")
public class ClaimsPageAPITest extends BaseApiTest { ... }

@ReportName("Admin Portal API Tests")  
public class AdminPortalTest extends BaseApiTest { ... }



@Override
public void onTestStart(ITestResult result) {
    Class<?> clazz = result.getTestClass().getRealClass();
    
    ReportName annotation = clazz.getAnnotation(ReportName.class);
    String parentName = (annotation != null) 
            ? annotation.value() 
            : clazz.getSimpleName();

    ExtentTest parent = classTests.computeIfAbsent(parentName,
            name -> ExtentManager.getInstance().createTest(name));
    
    // ... rest same as before
}
