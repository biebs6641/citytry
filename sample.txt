@Override
public void onStart(ITestContext context) {
    ExtentManager.getInstance();

    // Pre-create parent nodes in testng.xml order so the report follows that order
    for (org.testng.xml.XmlClass xmlClass : context.getCurrentXmlTest().getXmlClasses()) {
        try {
            Class<?> clazz = Class.forName(xmlClass.getName());
            ReportName annotation = clazz.getAnnotation(ReportName.class);
            String parentName = (annotation != null && !annotation.value().isEmpty())
                    ? annotation.value()
                    : clazz.getSimpleName();

            classTests.computeIfAbsent(parentName,
                    name -> ExtentManager.getInstance().createTest(name));
        } catch (ClassNotFoundException e) {
            System.err.println(">>> Could not load class for pre-registration: " 
                    + xmlClass.getName());
        }
    }
}
