import static org.hamcrest.MatcherAssert.assertThat;
import static io.restassured.module.jsv.JsonSchemaValidator.matchesJsonSchema;
import java.io.File;
import java.nio.file.Files;
import org.junit.Test;

public class SchemaValidationTest {

    @Test
    public void validateInputAgainstSchema() throws Exception {

        String inputJson = new String(Files.readAllBytes(
            new File("src/test/resources/input.json").toPath()
        ));

        assertThat(inputJson, matchesJsonSchema(
            new File("src/test/resources/schema.json")
        ));
    }
}
